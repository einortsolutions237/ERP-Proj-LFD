- [ ] **Step 1: Replace the hand-rolled `<button>` with `Button`**

In `src/components/reports/DownloadCsvButton.tsx`, change the whole file to:

```tsx
'use client'
import Button from '@/components/ui/Button'

export default function DownloadCsvButton({ filename, csv }: { filename: string; csv: string }) {
  function handleClick() {
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = filename
    a.click()
    URL.revokeObjectURL(url)
  }
  return (
    <Button type="button" variant="secondary" onClick={handleClick}>
      Download CSV
    </Button>
  )
}
```

The `handleClick` blob-download logic is byte-unchanged — only the rendered element changes, from a raw `<button className="...">` to `<Button variant="secondary">`.
